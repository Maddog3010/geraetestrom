In Home Assistant füllt sich die Liste automatisch mit allen Leistungssensoren (W).
1. configuration.yaml
input_select:
  geraet_sensor:   {name: Gerät – Leistung, options: ["-"]}
  solar_sensor:    {name: Solar – Leistung, options: ["-"]}
  batterie_sensor: {name: Batterie – Entladung, options: ["-"]}
  netz_sensor:     {name: Netz – Bezug, options: ["-"]}
  bezug_modus:     {name: Bezugspreis – Quelle, options: [Manuell, Sensor], initial: Manuell}
  bezug_sensor:    {name: Bezugspreis – Sensor, options: ["-"]}
  einspeise_modus: {name: Einspeisevergütung – Quelle, options: [Manuell, Sensor], initial: Manuell}
  einspeise_sensor: {name: Einspeisevergütung – Sensor, options: ["-"]}

input_number:
  strompreis:
    {name: Bezugspreis, min: 0, max: 1, step: 0.01, initial: 0.32, mode: box, unit_of_measurement: "€/kWh"}
  einspeise:
    {name: Einspeisevergütung, min: 0, max: 1, step: 0.01, initial: 0.08, mode: box, unit_of_measurement: "€/kWh"}

automation:
  - alias: Leistungssensoren zur Auswahl laden
    trigger: [{platform: homeassistant, event: start}]
    action:
      - repeat:
          for_each: [geraet_sensor, solar_sensor, batterie_sensor, netz_sensor]
          sequence:
            - service: input_select.set_options
              target: {entity_id: "input_select.{{ repeat.item }}"}
              data:
                options: >
                  {{ states.sensor | selectattr('attributes.device_class','eq','power')
                     | map(attribute='entity_id') | list }}

  - alias: Preissensoren zur Auswahl laden
    trigger: [{platform: homeassistant, event: start}]
    action:
      - repeat:
          for_each: [bezug_sensor, einspeise_sensor]
          sequence:
            - service: input_select.set_options
              target: {entity_id: "input_select.{{ repeat.item }}"}
              data:
                options: >
                  {{ states.sensor | selectattr('attributes.unit_of_measurement','defined')
                     | selectattr('attributes.unit_of_measurement','search','/kWh')
                     | map(attribute='entity_id') | list }}

template:
  - sensor:
      - {name: Gerät Solar-Anteil,    unique_id: geraet_solar_anteil,    unit_of_measurement: W, device_class: power, state: "{% from 'geraet.jinja' import anteil %}{{ anteil('solar')|trim }}"}
      - {name: Gerät Batterie-Anteil, unique_id: geraet_batterie_anteil, unit_of_measurement: W, device_class: power, state: "{% from 'geraet.jinja' import anteil %}{{ anteil('batterie')|trim }}"}
      - {name: Gerät Netz-Anteil,     unique_id: geraet_netz_anteil,     unit_of_measurement: W, device_class: power, state: "{% from 'geraet.jinja' import anteil %}{{ anteil('netz')|trim }}"}
      - name: Gerät Bezugspreis
        unique_id: geraet_bezugspreis
        unit_of_measurement: "€/kWh"
        state: >
          {{ states(states('input_select.bezug_sensor'))|float(0)
             if is_state('input_select.bezug_modus', 'Sensor')
             else states('input_number.strompreis')|float(0) }}
      - name: Gerät Einspeisepreis
        unique_id: geraet_einspeisepreis
        unit_of_measurement: "€/kWh"
        state: >
          {{ states(states('input_select.einspeise_sensor'))|float(0)
             if is_state('input_select.einspeise_modus', 'Sensor')
             else states('input_number.einspeise')|float(0) }}
      - name: Gerät Kosten heute
        unit_of_measurement: "€"
        state: "{{ (states('sensor.geraet_netz_heute')|float(0) * states('sensor.geraet_bezugspreis')|float(0))|round(2) }}"
      - name: Gerät Ersparnis heute
        unit_of_measurement: "€"
        state: >
          {% set p = states('sensor.geraet_bezugspreis')|float(0) %}
          {% set f = states('sensor.geraet_einspeisepreis')|float(0) %}
          {{ (states('sensor.geraet_solar_heute')|float(0) * (p - f)
            + states('sensor.geraet_batterie_heute')|float(0) * p)|round(2) }}
      - name: Gerät Eigenversorgung
        unit_of_measurement: "%"
        state: >
          {% set e = states('sensor.geraet_solar_heute')|float(0) + states('sensor.geraet_batterie_heute')|float(0) %}
          {% set t = e + states('sensor.geraet_netz_heute')|float(0) %}
          {{ (100 * e / t)|round(0) if t > 0 else 0 }}

sensor:
  - {platform: integration, source: sensor.geraet_solar_anteil,    name: Gerät Solar kWh,    unit_prefix: k, round: 3}
  - {platform: integration, source: sensor.geraet_batterie_anteil, name: Gerät Batterie kWh, unit_prefix: k, round: 3}
  - {platform: integration, source: sensor.geraet_netz_anteil,     name: Gerät Netz kWh,     unit_prefix: k, round: 3}

utility_meter:
  geraet_solar_heute:    {source: sensor.geraet_solar_kwh,    cycle: daily}
  geraet_batterie_heute: {source: sensor.geraet_batterie_kwh, cycle: daily}
  geraet_netz_heute:     {source: sensor.geraet_netz_kwh,     cycle: daily}
2. custom_templates/geraet.jinja
{% macro anteil(quelle) %}
{% set g = states(states('input_select.geraet_sensor'))|float(0) %}
{% set s = [states(states('input_select.solar_sensor'))|float(0), 0]|max %}
{% set b = [states(states('input_select.batterie_sensor'))|float(0), 0]|max %}
{% set n = [states(states('input_select.netz_sensor'))|float(0), 0]|max %}
{% set t = s + b + n %}
{% set w = {'solar': s, 'batterie': b, 'netz': n}[quelle] %}
{{ (g * w / t)|round(1) if t > 0 else 0 }}
{% endmacro %}

          - {type: tile, entity: sensor.geraet_netz_anteil,     name: Netz,     icon: "mdi:transmission-tower",   color: indigo}
      - type: gauge
        entity: sensor.geraet_eigenversorgung
        name: Eigener Strom
        min: 0
        max: 100
        severity: {red: 0, yellow: 30, green: 60}
      - type: history-graph
        title: Herkunft des Stroms
        hours_to_show: 24
        entities:
          - sensor.geraet_solar_anteil
          - sensor.geraet_batterie_anteil
          - sensor.geraet_netz_anteil
      - type: entities
        title: Kosten & Ersparnis heute
        entities:
          - sensor.geraet_kosten_heute
          - sensor.geraet_ersparnis_heute
      - type: entities
        title: Sensoren & Preise
        entities:
          - input_select.geraet_sensor
          - input_select.solar_sensor
          - input_select.batterie_sensor
          - input_select.netz_sensor
          - input_select.bezug_modus
          - input_number.strompreis
